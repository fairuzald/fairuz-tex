% This flag is disabled when the appendices begin.  It lets ordinary tables in
% the report stay together while allowing intentionally long appendix lists to
% split naturally across pages.
\newif\ifmainreporttables
\mainreporttablesfalse
% Tables use a smaller type size in their local environments, but their rows
% remain single-spaced as required by the report-writing guidelines.
\AtBeginEnvironment{longtable}{\singlespacing}
% Do not carry the document-wide paragraph-fill stretch into split table
% cells, it can produce infinite-glue diagnostics when a longtable is broken
% across pages.
\AtBeginEnvironment{longtable}{\setlength{\parfillskip}{0pt}}
% Keep ordinary-sized tables in the main chapters together when they can fit
% on one page.  The question lists and any intentionally detailed appendix
% material are allowed to split naturally without that page reservation.
\AtBeginEnvironment{longtable}{%
    \ifmainreporttables
        \Needspace{0.58\textheight}%
    \fi
}
